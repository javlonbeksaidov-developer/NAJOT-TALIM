cars = ['malibu', 'BMW', 'mercedes', 21, 49, 77]

# print(cars[-1])

# print(cars[3:])
# print(cars[:])
# print(cars[::-1])

# ma'lumotlarni o'zgartirish
cars[3] = 'tayoto'
cars[4] = 'matiz'
cars[5] = 'cobalt'

# append() metodi -> element qo'shish
cars.append('spark')

# insert() metodi -> elementni index bo'yicha qo'shadi
cars.insert(2, 'captiva')

# remove() metodi -> element o'chirish
cars.remove("malibu")

# pop() metodi -> elementni index bo'yicha sug'irib oladi va o'zgaruvchiga yuklaydi
pop_item = cars.pop(-1)
print(pop_item)

# extend() metodi -> kengaytirish
# cars.extend('javlon')
cars.extend(['nexia', 'porj'])

print(cars)

# reverse() metodi
cars.reverse()
print(cars)

# sort() metodi
cars.sort()
print(cars)

cars.sort(reverse=True)
print(cars)
