# calculator



def yigindi(a, b):
    return a + b

def ayirma(a, b):
    return a - b

def kopaytma(a, b):
    return a * b

def bolinma(a, b):
    try:
        return a / b
    except ZeroDivisionError:
        return "Nolga bo'linmaydi."

def main():
    print("=== Calculator ===\n")
    while True:
        try:
            son_1 = int(input("Birinchi son: "))
        except ValueError:
            print("Noto'g'ri qiymat")

        while True:
            amal = input("Amal (+) (-) (*) (/): ")

            if amal not in "+-/*":
                print("Noto'g'ri amal!\n")
                continue
            else:
                break

        try:            
            son_2 = int(input("Ikkinchi son: "))
        except ValueError:
            print("Noto'g'ri qiymat")
        
        if amal == "+":
            result = yigindi(son_1, son_2)
        elif amal == "-":
            result = ayirma(son_1, son_2)
        elif amal == "/":
            result = bolinma(son_1, son_2)
        elif amal == "*":
            result = kopaytma(son_1, son_2)
        else:
            print("Noto'g'ri amal")

        print(f"Natija: {son_1} {amal} {son_2} = {result}")

        yana = input("\nDavom ettirasizmi? (yes/no)").strip().lower()
        if yana == "no":
            break


if __name__ == "__main__":
    main()
