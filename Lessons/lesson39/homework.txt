# 3) Bank uchun customers va accounts jadvallarini yarating. 
# Customer UUID orqali aniqlansin. Account uchun alohida sequence ishlating. 
# Har bir customer bir nechta accountga ega bo‘lishi mumkin bo‘lgan 
# struktura yarating.

>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>

CREATE EXTENSION IF NOT EXISTS pgcrypto;

CREATE TABLE customers (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    full_name VARCHAR(100) NOT NULL,
    phone VARCHAR(20),
    email VARCHAR(100) UNIQUE
);

>>>>>>>>>>>>>>

CREATE SEQUENCE account_id_seq
START 1000
INCREMENT 1;

>>>>>>>>>>>>>>

CREATE TABLE accounts (
    id INTEGER PRIMARY KEY DEFAULT nextval('account_id_seq'),
    customer_id UUID NOT NULL,
    account_number VARCHAR(30) UNIQUE NOT NULL,
    balance NUMERIC(15,2) DEFAULT 0,
    account_type VARCHAR(30),

    CONSTRAINT fk_customer
        FOREIGN KEY (customer_id)
        REFERENCES customers(id)
);
